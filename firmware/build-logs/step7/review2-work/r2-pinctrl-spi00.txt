	/* storage card (design/firmware.md 5.1). Default: SCK and MOSI driven (low when idle). Sleep: SD_SCK, SD_MOSI and

	 * off (the pull-ups R60, R63 to R65 go to VDD_SD). The fourth line, SD_CS, is a GPIO (cs-gpios of spi00) and in no
	 * pin state: the kit's SPI driver keeps it an output; modules/ring_store/src/rst_hw_real.c disconnects it after each

	};

		};
	};
