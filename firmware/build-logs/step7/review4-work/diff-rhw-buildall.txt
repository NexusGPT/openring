+	python3 - "$1" <<'PYCHECK'
+import re, sys
+src = open(sys.argv[1]).read()
+code = re.sub(r"/\*.*?\*/", " ", src, flags=re.S)
+def body(name):

+    return " ".join(m.group(1).split()) if m else None
