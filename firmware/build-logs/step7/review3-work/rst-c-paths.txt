   341	static int card_up(void)
   342	{
   343		int r = -EIO;
   344	
   345		if (rhw_pmic()->rail_set(RHW_RAIL_SD, true, "rst_mount") != 0) {                  /* FW l.506: rail on */
   346			return -EIO;
   347		}
   348		k_msleep(RREC_SD_RAIL_WAIT_MS);                                                  /* wait 1 ms */
   349		(void)rhw_sdbus()->resume();                                                     /* resume spi00 */
   350		for (uint32_t i = 0; i < RREC_INIT_TRIES; i++) {                                 /* FW l.138 */
   351			r = rhw_disk()->init();
   352			if (r == 0) {
   353				break;
   354			}
   355			if (i + 1 < RREC_INIT_TRIES) {
   356				k_msleep(RREC_INIT_GAP_MS);
   357			}
   358		}
   359		if (r != 0) {
   360			power_down(false, "rst_mount");
   361			return -EIO;
   362		}
   363		return 0;
   364	}
