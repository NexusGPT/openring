+ * Ring devices: the device list and the switch decision. Pure logic: no
+ * kernel, driver or Bluetooth call; runs on native_sim with no radio.
+ *
+ * Three slots. Slot n is local Bluetooth identity n of the ring (one identity
+ * per paired device, as nRF Desktop does: ble_bond.rst lines 48, 54, 57-58).
+ * A slot is paired when a bond exists for its identity; the Bluetooth part

+#define RDEV_SLOTS 3
+#define RDEV_NONE  0xFF   /* no slot */
+#define RDEV_ANY   0xFE   /* a request for any free slot */
+
+struct rdev_list {
+	uint8_t paired;  /* bit n set: slot n holds a pairing */
+	uint8_t active;  /* the active slot, or RDEV_NONE */
+};
+
+void rdev_list_init(struct rdev_list *l);
+bool rdev_list_is_paired(const struct rdev_list *l, uint8_t slot);
+uint8_t rdev_list_count(const struct rdev_list *l);
+/* The lowest empty slot, or RDEV_NONE when all three hold a pairing. */
+uint8_t rdev_list_free_slot(const struct rdev_list *l);
+
+/* Slot to use for a new pairing: requested is a slot number or RDEV_ANY.
+ * Returns the slot, or a negative errno: -EINVAL out of range, -EEXIST the
+ * requested slot is paired (remove it first), -ENOSPC no free slot. */
+int rdev_list_choose_pair_slot(const struct rdev_list *l, uint8_t requested);
+
+/* A new pairing was made in the slot: it is paired; it becomes active only when no slot is active.
+ * -EINVAL out of range, -EEXIST already paired. */
+int rdev_list_add(struct rdev_list *l, uint8_t slot);
+
+/* The pairing of the slot was removed. When it was the active one, the next
+ * paired slot after it (in slot order, wrapping) becomes active, or none.
+ * -EINVAL out of range, -ENOENT not paired. */
+int rdev_list_remove(struct rdev_list *l, uint8_t slot);
+
+/* The paired slot after from, wrapping, other than from; RDEV_NONE if none. */
+uint8_t rdev_list_next(const struct rdev_list *l, uint8_t from);

+/* Ring devices: pure logic (device list, switch decision, plan). */
+
+#include <errno.h>
+
+#include <ring_devices/rdev_logic.h>
+
+void rdev_list_init(struct rdev_list *l)
+{
+	l->paired = 0;
+	l->active = RDEV_NONE;
+}
+
+bool rdev_list_is_paired(const struct rdev_list *l, uint8_t slot)
+{
+	return slot < RDEV_SLOTS && (l->paired & (1U << slot)) != 0;
+}
+
+uint8_t rdev_list_count(const struct rdev_list *l)
+{
+	uint8_t n = 0;
+
+	for (uint8_t i = 0; i < RDEV_SLOTS; i++) {
+		n += rdev_list_is_paired(l, i) ? 1 : 0;
+	}
+	return n;
+}
+
+uint8_t rdev_list_free_slot(const struct rdev_list *l)
+{
+	for (uint8_t i = 0; i < RDEV_SLOTS; i++) {
+		if (!rdev_list_is_paired(l, i)) {
+			return i;
+		}
+	}
+	return RDEV_NONE;
+}
+
+int rdev_list_choose_pair_slot(const struct rdev_list *l, uint8_t requested)
+{
+	if (requested == RDEV_ANY) {
+		uint8_t s = rdev_list_free_slot(l);
+
+		return s == RDEV_NONE ? -ENOSPC : (int)s;
+	}
+	if (requested >= RDEV_SLOTS) {
+		return -EINVAL;
+	}
+	if (rdev_list_is_paired(l, requested)) {
+		return -EEXIST;
+	}
+	return requested;
+}
+
+int rdev_list_add(struct rdev_list *l, uint8_t slot)
+{
+	if (slot >= RDEV_SLOTS) {
+		return -EINVAL;
+	}
+	if (rdev_list_is_paired(l, slot)) {
+		return -EEXIST;
+	}
+	l->paired |= (uint8_t)(1U << slot);
+	/* the new device becomes the active one only when no device is active (design/firmware.md 3.4, 7.4) */
+	if (l->active == RDEV_NONE) {
+		l->active = slot;
+	}
+	return 0;
+}
+
+uint8_t rdev_list_next(const struct rdev_list *l, uint8_t from)
+{
+	for (uint8_t k = 1; k <= RDEV_SLOTS; k++) {
+		uint8_t s = (uint8_t)((from + k) % RDEV_SLOTS);
+
+		if (s != from && rdev_list_is_paired(l, s)) {
+			return s;
+		}
+	}
+	return RDEV_NONE;
+}
+
+int rdev_list_remove(struct rdev_list *l, uint8_t slot)
+{
+	if (slot >= RDEV_SLOTS) {
+		return -EINVAL;
+	}
+	if (!rdev_list_is_paired(l, slot)) {
+		return -ENOENT;
+	}
+	l->paired &= (uint8_t)~(1U << slot);
+	if (l->active == slot) {
+		l->active = rdev_list_next(l, slot);
+	}
+	return 0;
+}
+
+enum rdev_switch rdev_list_decide_switch(const struct rdev_list *l, uint8_t target)
+{
+	if (target >= RDEV_SLOTS) {
+		return RDEV_SWITCH_REFUSE_RANGE;
+	}
+	if (!rdev_list_is_paired(l, target)) {
+		return RDEV_SWITCH_REFUSE_EMPTY;
+	}
+	if (l->active == target) {
+		return RDEV_SWITCH_NOP;
+	}
+	return RDEV_SWITCH_GO;
+}
+
+enum rdev_switch rdev_list_switch(struct rdev_list *l, uint8_t target)
+{
+	enum rdev_switch d = rdev_list_decide_switch(l, target);
+
+	if (d == RDEV_SWITCH_GO) {
+		l->active = target;
+	}
+	return d;
+}
+
+struct rdev_plan rdev_plan_next(const struct rdev_state *s)
+{
+	struct rdev_plan p = { RDEV_STEP_IDLE, RDEV_NONE, false };
+	uint8_t target = s->window ? s->window_slot : s->active;
+
+	if (target >= RDEV_SLOTS) {
+		/* nothing to be connected to: end any link */
+		p.step = s->link_slot != RDEV_NONE ? RDEV_STEP_DISCONNECT : RDEV_STEP_IDLE;
+		return p;
+	}
+	if (s->link_slot != RDEV_NONE) {
+		p.step = s->link_slot == target ? RDEV_STEP_STAY : RDEV_STEP_DISCONNECT;
+		return p;
+	}
+	p.step = RDEV_STEP_ADVERTISE;
+	p.slot = target;
+	p.open = s->window;
+	return p;
+}
+
+const char *rdev_step_name(enum rdev_step s)
+{
+	switch (s) {
+	case RDEV_STEP_IDLE:
+		return "idle";
+	case RDEV_STEP_STAY:
+		return "stay";
+	case RDEV_STEP_DISCONNECT:
+		return "disconnect";
+	case RDEV_STEP_ADVERTISE:
+		return "advertise";
+	}
+	return "?";
+}
+
+bool rdev_link_usable(uint8_t link_slot, bool secure, uint8_t active)
+{
+	return secure && link_slot != RDEV_NONE && link_slot == active;
+}

+	       RREC_CHUNK_PAYLOAD_BYTES - (size_t)h->frames * RREC_CHANNELS * sizeof(int16_t));
+	pcrc = crc32_ieee(payload, RREC_CHUNK_PAYLOAD_BYTES);
+
+	memset(o, 0, RREC_CHUNK_HEADER_BYTES);
+	rst_put32(o + RREC_CH_OFF_MAGIC, RREC_CHUNK_MAGIC);
+	rst_put16(o + RREC_CH_OFF_VERSION, RREC_CHUNK_HEADER_VERSION);
+	rst_put16(o + RREC_CH_OFF_SIZE, RREC_CHUNK_HEADER_BYTES);
+	rst_put32(o + RREC_CH_OFF_CHUNK_SEQ, h->chunk_seq);
+	rst_put32(o + RREC_CH_OFF_RECORDING_ID, h->recording_id);
+	rst_put32(o + RREC_CH_OFF_CHUNK_IN_REC, h->chunk_in_recording);
+	rst_put16(o + RREC_CH_OFF_FLAGS, h->flags);
+	o[RREC_CH_OFF_RECORDING_TYPE] = h->recording_type;
+	o[RREC_CH_OFF_CHANNELS] = RREC_CHANNELS;
+	rst_put32(o + RREC_CH_OFF_SAMPLE_RATE, RREC_PCM_RATE_HZ);
+	o[RREC_CH_OFF_SAMPLE_FORMAT] = RREC_SAMPLE_FORMAT_S16LE;
+	o[RREC_CH_OFF_MIC_MASK] = RREC_MIC_MASK;
+	rst_put32(o + RREC_CH_OFF_FRAMES, h->frames);
+	rst_put32(o + RREC_CH_OFF_PAYLOAD_BYTES, h->frames * RREC_CHANNELS * (uint32_t)sizeof(int16_t));
+	rst_put64(o + RREC_CH_OFF_FIRST_FRAME_INDEX, h->first_frame_index);
+	rst_put64(o + RREC_CH_OFF_UTC_MS, (uint64_t)h->utc_ms);
+	rst_put64(o + RREC_CH_OFF_UPTIME_NS, h->uptime_ns);
+	rst_put32(o + RREC_CH_OFF_PDM_CLOCK_HZ, RREC_PDM_CLOCK_HZ);
+	rst_put16(o + RREC_CH_OFF_PDM_RATIO, RREC_PDM_RATIO);
+	o[RREC_CH_OFF_GAIN_CODE] = RREC_PDM_GAIN_CODE;
+	rst_put16(o + RREC_CH_OFF_HIGHLIGHT_COUNT, h->highlight_count);
+	for (uint32_t i = 0; i < RREC_HIGHLIGHTS_PER_CHUNK; i++) {
+		rst_put32(o + RREC_CH_OFF_HIGHLIGHT_FRAME + i * sizeof(uint32_t),
+			  i < h->highlight_count ? h->highlight_frame[i] : RREC_UNUSED_U32);
+	}
+	rst_put16(o + RREC_CH_OFF_BATTERY_MV, h->battery_mv);
+	rst_put16(o + RREC_CH_OFF_TEMP_C_X10, (uint16_t)h->temp_c_x10);
+	rst_put32(o + RREC_CH_OFF_BOOT_COUNT, h->boot_count);
+	rst_put32(o + RREC_CH_OFF_DROPOUT_BLOCKS, h->dropout_blocks);
+	rst_put32(o + RREC_CH_OFF_FW_VERSION, h->fw_version);
+	rst_put32(o + RREC_CH_OFF_CYCLE_HZ, h->cycle_hz);
+	rst_put16(o + RREC_CH_OFF_MOTOR_COUNT, h->motor_count);
+	for (uint32_t i = 0; i < RREC_MOTOR_PER_CHUNK; i++) {
+		uint8_t *p = o + RREC_CH_OFF_MOTOR_FRAMES + i * 2 * sizeof(uint32_t);
+
+		rst_put32(p, i < h->motor_count ? h->motor_frames[i][0] : RREC_UNUSED_U32);
+		rst_put32(p + sizeof(uint32_t), i < h->motor_count ? h->motor_frames[i][1] : RREC_UNUSED_U32);
+	}
+	rst_put32(o + RREC_CH_OFF_PAYLOAD_CRC, pcrc);
+	rst_put32(o + RREC_CH_OFF_HEADER_CRC, chunk_hdr_crc(o));
+	if (payload_crc != NULL) {
+		*payload_crc = pcrc;
+	}
+	return 0;
+}

+int rst_fmt_chunk_hdr_decode(const uint8_t *in, struct rst_chunk_hdr *h, uint32_t *payload_crc)
+{
+	if (rst_get32(in + RREC_CH_OFF_MAGIC) != RREC_CHUNK_MAGIC ||
+	    rst_get16(in + RREC_CH_OFF_VERSION) != RREC_CHUNK_HEADER_VERSION ||
+	    rst_get16(in + RREC_CH_OFF_SIZE) != RREC_CHUNK_HEADER_BYTES ||
+	    rst_get32(in + RREC_CH_OFF_HEADER_CRC) != chunk_hdr_crc(in)) {
+		return -EBADMSG;
+	}
+	h->chunk_seq = rst_get32(in + RREC_CH_OFF_CHUNK_SEQ);
+	h->recording_id = rst_get32(in + RREC_CH_OFF_RECORDING_ID);
+	h->chunk_in_recording = rst_get32(in + RREC_CH_OFF_CHUNK_IN_REC);
+	h->flags = rst_get16(in + RREC_CH_OFF_FLAGS);
+	h->recording_type = in[RREC_CH_OFF_RECORDING_TYPE];
+	h->frames = rst_get32(in + RREC_CH_OFF_FRAMES);
+	h->first_frame_index = rst_get64(in + RREC_CH_OFF_FIRST_FRAME_INDEX);
+	h->utc_ms = (int64_t)rst_get64(in + RREC_CH_OFF_UTC_MS);
+	h->uptime_ns = rst_get64(in + RREC_CH_OFF_UPTIME_NS);
+	h->highlight_count = rst_get16(in + RREC_CH_OFF_HIGHLIGHT_COUNT);
+	for (uint32_t i = 0; i < RREC_HIGHLIGHTS_PER_CHUNK; i++) {
+		h->highlight_frame[i] = rst_get32(in + RREC_CH_OFF_HIGHLIGHT_FRAME + i * sizeof(uint32_t));
+	}
+	h->battery_mv = rst_get16(in + RREC_CH_OFF_BATTERY_MV);
+	h->temp_c_x10 = (int16_t)rst_get16(in + RREC_CH_OFF_TEMP_C_X10);
+	h->boot_count = rst_get32(in + RREC_CH_OFF_BOOT_COUNT);
+	h->dropout_blocks = rst_get32(in + RREC_CH_OFF_DROPOUT_BLOCKS);
+	h->fw_version = rst_get32(in + RREC_CH_OFF_FW_VERSION);
+	h->cycle_hz = rst_get32(in + RREC_CH_OFF_CYCLE_HZ);
+	h->motor_count = rst_get16(in + RREC_CH_OFF_MOTOR_COUNT);
+	for (uint32_t i = 0; i < RREC_MOTOR_PER_CHUNK; i++) {
+		const uint8_t *p = in + RREC_CH_OFF_MOTOR_FRAMES + i * 2 * sizeof(uint32_t);
+
+		h->motor_frames[i][0] = rst_get32(p);
+		h->motor_frames[i][1] = rst_get32(p + sizeof(uint32_t));
+	}
+	if (payload_crc != NULL) {
+		*payload_crc = rst_get32(in + RREC_CH_OFF_PAYLOAD_CRC);
+	}
+	return 0;
+}
+
+int rst_fmt_chunk_verify(const uint8_t *chunk, struct rst_chunk_hdr *h, uint32_t *payload_crc)
+{
+	uint32_t pcrc;
+	int r = rst_fmt_chunk_hdr_decode(chunk, h, &pcrc);
+
+	if (r != 0) {
+		return r;
+	}
+	if (crc32_ieee(chunk + RREC_CHUNK_HEADER_BYTES, RREC_CHUNK_PAYLOAD_BYTES) != pcrc) {
+		return -EILSEQ;
+	}
+	if (payload_crc != NULL) {
+		*payload_crc = pcrc;
+	}
+	return 0;
+}

+void rst_fmt_log_encode(const struct rrec_logrec *r, uint8_t *out)
+{
+	memset(out, 0, RREC_LOG_RECORD_BYTES);
+	rst_put32(out + RREC_LG_OFF_MAGIC, RREC_LOG_MAGIC);
+	rst_put32(out + RREC_LG_OFF_SEQ, r->seq);
+	rst_put64(out + RREC_LG_OFF_UPTIME_NS, (uint64_t)r->uptime_ns);
+	rst_put64(out + RREC_LG_OFF_UTC_MS, (uint64_t)r->utc_ms);
+	rst_put16(out + RREC_LG_OFF_CODE, r->code);
+	out[RREC_LG_OFF_STATE] = r->state;
+	memcpy(out + RREC_LG_OFF_ARGS, r->args, RREC_LOG_ARG_BYTES);
+	if (r->has_crash) {
+		uint8_t *p = out + RREC_LG_OFF_CRASH;
+
+		rst_put32(p, r->crash.reason);
+		rst_put32(p + 4, r->crash.pc);
+		rst_put32(p + 8, r->crash.lr);
+		rst_put32(p + 12, r->crash.psr);
+		p += 16;
+		memcpy(p, r->crash.thread, RREC_CRASH_NAME_BYTES);
+		p += RREC_CRASH_NAME_BYTES;
+		for (uint32_t i = 0; i < RREC_CRASH_STACK_WORDS; i++) {
+			rst_put32(p + i * sizeof(uint32_t), r->crash.stack[i]);
+		}
+	}
+	rst_put32(out + RREC_LG_OFF_CRC, crc32_ieee(out, RREC_LG_OFF_CRC));
+}

+int rst_fmt_log_decode(const uint8_t *in, struct rrec_logrec *r)
+{
+	if (rst_get32(in + RREC_LG_OFF_MAGIC) != RREC_LOG_MAGIC ||
+	    rst_get32(in + RREC_LG_OFF_CRC) != crc32_ieee(in, RREC_LG_OFF_CRC)) {
+		return -EBADMSG;
+	}
+	memset(r, 0, sizeof(*r));
+	r->seq = rst_get32(in + RREC_LG_OFF_SEQ);
+	r->uptime_ns = (int64_t)rst_get64(in + RREC_LG_OFF_UPTIME_NS);
+	r->utc_ms = (int64_t)rst_get64(in + RREC_LG_OFF_UTC_MS);
+	r->code = rst_get16(in + RREC_LG_OFF_CODE);
+	r->state = in[RREC_LG_OFF_STATE];
+	memcpy(r->args, in + RREC_LG_OFF_ARGS, RREC_LOG_ARG_BYTES);
+	const uint8_t *p = in + RREC_LG_OFF_CRASH;
+
+	for (uint32_t i = 0; i < RREC_CRASH_BLOCK_BYTES; i++) {
+		if (p[i] != 0) {
+			r->has_crash = true;
+			break;
+		}
+	}
+	if (r->has_crash) {
+		r->crash.reason = rst_get32(p);
+		r->crash.pc = rst_get32(p + 4);
+		r->crash.lr = rst_get32(p + 8);
+		r->crash.psr = rst_get32(p + 12);
+		p += 16;
+		memcpy(r->crash.thread, p, RREC_CRASH_NAME_BYTES);
+		p += RREC_CRASH_NAME_BYTES;
+		for (uint32_t i = 0; i < RREC_CRASH_STACK_WORDS; i++) {
+			r->crash.stack[i] = rst_get32(p + i * sizeof(uint32_t));
+		}
+	}
+	return 0;
+}

+{

+	}

+void rst_recs_apply(struct rst_recs *t, const struct rst_ix *x)
+{
+	struct rst_rec *r;
+
+	switch (x->type) {
+	case RREC_IX_TYPE_CLOSE:
+		if (t->first_id == 0) {
+			t->first_id = x->recording_id;
+		}
+		if (x->recording_id < t->first_id || rst_recs_find(t, x->recording_id) != NULL) {
+			return;       /* a record this table already has or has seen the end of */
+		}
+		if (t->n == t->cap) {
+			t->evict_max_id = t->ent[0].info.recording_id;
+			memmove(&t->ent[0], &t->ent[1], (t->n - 1) * sizeof(t->ent[0]));
+			t->n--;
+		}
+		rec_from_ix(&t->ent[t->n++], x);
+		t->live_total++;
+		break;
+	case RREC_IX_TYPE_CONFIRM:
+		r = rst_recs_find(t, x->recording_id);
+		if (r != NULL) {
+			rec_confirm(r);
+		}
+		break;
+	case RREC_IX_TYPE_DELETE:
+		r = rst_recs_find(t, x->recording_id);
+		if (r != NULL) {
+			uint32_t i = (uint32_t)(r - t->ent);
+
+			memmove(&t->ent[i], &t->ent[i + 1], (t->n - i - 1) * sizeof(t->ent[0]));
+			t->n--;
+			t->live_total--;
+		} else if (t->first_id != 0 && x->recording_id >= t->first_id && x->recording_id <= t->evict_max_id &&
+			   t->live_total > 0) {
+			t->live_total--;       /* a live recording that was pushed out of the table */
+		}
+		break;
+	default:
+		break;
+	}
+}

+	return 0;

+		if (r != 0) {
+			return r;
+		}
+		if (x.recording_id < id_lo || x.recording_id - id_lo >= win) {
+			continue;
+		}
+		struct rst_rec *slot = &out[x.recording_id - id_lo];
+
+		if (x.type == RREC_IX_TYPE_CLOSE) {
+			rec_from_ix(slot, &x);
+		} else if (x.type == RREC_IX_TYPE_CONFIRM && slot->info.recording_id != 0) {
+			rec_confirm(slot);
+		} else if (x.type == RREC_IX_TYPE_DELETE) {
+			memset(slot, 0, sizeof(*slot));
+		}
+	}
+	uint32_t m = 0;
+
+	for (uint32_t i = 0; i < win; i++) {
+		if (out[i].info.recording_id != 0) {
+			if (m != i) {
+				out[m] = out[i];
+			}
+			m++;
+		}
+	}
+	*n = m;

+int rst_log_tail_walk(rst_rd_fn rd, void *ctx, uint32_t index_tail, uint32_t index_head, uint32_t head_chunk_seq,

+
+	for (uint32_t n = index_tail; n != index_head; n++) {
+		int r = read_ix(rd, ctx, n, &c, sector);

+		if (r != 0) {
+			return r;
+		}
+		if (c.type != RREC_IX_TYPE_CLOSE) {
+			continue;
+		}
+		bool deleted = false;
+
+		for (uint32_t m = n + 1; m != index_head && !deleted; m++) {
+			r = read_ix(rd, ctx, m, &x, sector);
+			if (r != 0) {
+				return r;
+			}
+			deleted = x.type == RREC_IX_TYPE_DELETE && x.recording_id == c.recording_id;
+		}
+		if (!deleted) {
+			*new_chunk_tail = c.first_chunk_seq;
+			*new_index_tail = n;
+			return 0;
+		}
+	}
+	*new_chunk_tail = head_chunk_seq;
+	*new_index_tail = index_head;
+	return 0;

+int rst_log_scan(rst_rd_fn rd, void *rctx, uint32_t total, uint32_t recording_id, uint32_t first_seq, uint8_t *buf,
+		 void (*progress)(void *ctx, uint32_t kept), void *pctx, struct rst_scan *out)
+{
+	struct rst_chunk_hdr h;
+	uint32_t pcrc;
+
+	memset(out, 0, sizeof(*out));
+	while (out->kept < total) {
+		uint32_t seq = first_seq + out->kept;
+		uint32_t lba = rst_lay_chunk_lba(total, seq);
+
+		if (rd(rctx, lba, buf, RREC_CHUNK_SECTORS) != 0) {
+			out->err_lba = lba;
+			return -EIO;
+		}
+		if (rst_fmt_chunk_verify(buf, &h, &pcrc) != 0 || h.recording_id != recording_id || h.chunk_seq != seq) {
+			break;
+		}
+		if (out->kept == 0) {
+			out->start_utc_ms = h.utc_ms;
+			out->recording_type = h.recording_type;
+		}
+		out->digest = rst_fmt_digest_add(out->digest, pcrc);
+		out->duration_frames += h.frames;
+		for (uint32_t i = 0; i < h.highlight_count && i < RREC_HIGHLIGHTS_PER_CHUNK; i++) {
+			if (out->highlight_count < RREC_CLOSE_HIGHLIGHTS) {
+				out->highlight_ms[out->highlight_count] =
+					(uint32_t)(((uint64_t)h.highlight_frame[i] * 1000u) / RREC_PCM_RATE_HZ);   /* frames from the first kept frame (FW l.566) */
+			}
+			out->highlight_count++;
+		}
+		out->kept++;
+		if (progress != NULL) {
+			progress(pctx, out->kept);
+		}
+	}
+	for (uint32_t i = out->highlight_count; i < RREC_CLOSE_HIGHLIGHTS; i++) {
+		out->highlight_ms[i] = RREC_UNUSED_U32;
+	}
+	return 0;
+}

+#define DESC_TYPE_INTERFACE 0x04u   /* USB 2.0 table 9-5 */
+#define DESC_TYPE_ENDPOINT 0x05u    /* USB 2.0 table 9-5 */
+#define EP_ATTR_BULK 0x02u          /* USB 2.0 table 9-13 */
+
+uint16_t rxf_desc_mps(enum rhw_usb_speed speed)
+{
+	return speed == RHW_USB_HS ? (uint16_t)RREC_USB_MPS_HS : (uint16_t)RREC_USB_MPS_FS;
+}

+size_t rxf_desc_get(enum rhw_usb_speed speed, uint8_t out[RXF_DESC_BYTES])
+{
+	uint16_t mps = rxf_desc_mps(speed);
+	const uint8_t eps[2] = { RREC_USB_EP_OUT, RREC_USB_EP_IN };
+	uint8_t *p = out;
+
+	/* interface: length 9, type 4, number 0 (the stack assigns it), alternate 0, 2 endpoints, class, subclass,
+	 * protocol, no string */
+	*p++ = 9u;
+	*p++ = DESC_TYPE_INTERFACE;
+	*p++ = 0u;
+	*p++ = 0u;
+	*p++ = 2u;
+	*p++ = RREC_USB_CLASS;
+	*p++ = RREC_USB_SUBCLASS;
+	*p++ = RREC_USB_PROTOCOL;
+	*p++ = 0u;
+	for (int i = 0; i < 2; i++) {
+		*p++ = 7u;
+		*p++ = DESC_TYPE_ENDPOINT;
+		*p++ = eps[i];
+		*p++ = EP_ATTR_BULK;
+		*p++ = (uint8_t)mps;
+		*p++ = (uint8_t)(mps >> 8);
+		*p++ = 0u;                      /* bulk: no interval */
+	}
+	return (size_t)(p - out);
+}

+ * argument limits, response layouts, BUSY. No kernel call (decision D11a: crc32_ieee() of the C library is a library

+/* ------------------------------------------------------------------ little-endian helpers (FW l.70) */
+static uint16_t get16(const uint8_t *p)
+{
+	return (uint16_t)(p[0] | (p[1] << 8));
+}
+
+static uint32_t get32(const uint8_t *p)
+{
+	return (uint32_t)p[0] | ((uint32_t)p[1] << 8) | ((uint32_t)p[2] << 16) | ((uint32_t)p[3] << 24);
+}
+
+static uint64_t get64(const uint8_t *p)
+{
+	return (uint64_t)get32(p) | ((uint64_t)get32(p + 4) << 32);
+}

+static void put16(uint8_t *p, uint16_t v)
+{
+	p[0] = (uint8_t)v;
+	p[1] = (uint8_t)(v >> 8);
+}

+static void put32(uint8_t *p, uint32_t v)
+{
+	p[0] = (uint8_t)v;
+	p[1] = (uint8_t)(v >> 8);
+	p[2] = (uint8_t)(v >> 16);
+	p[3] = (uint8_t)(v >> 24);

+uint32_t rxf_proto_payload_crc(const uint8_t *payload, size_t length)

+void rxf_proto_put_header(uint8_t out[RREC_MSG_HEADER_BYTES], uint8_t opcode, enum rrec_status st, uint32_t tag,
+			  uint32_t length, uint32_t crc)
+{
+	put16(out + RREC_MSG_OFF_MAGIC, RREC_MSG_MAGIC);
+	out[RREC_MSG_OFF_OPCODE] = opcode;
+	out[RREC_MSG_OFF_STATUS] = (uint8_t)st;
+	put32(out + RREC_MSG_OFF_TAG, tag);
+	put32(out + RREC_MSG_OFF_LENGTH, length);
+	put32(out + RREC_MSG_OFF_CRC, crc);
+}

+enum rrec_status rxf_proto_check(const uint8_t *req, size_t req_len, struct rxf_req *h)
+{
+	uint32_t expect;
+
+	memset(h, 0, sizeof(*h));
+	if (req_len < RREC_MSG_HEADER_BYTES) {
+		return RREC_ST_BAD_MAGIC;
+	}
+	h->opcode = req[RREC_MSG_OFF_OPCODE];
+	h->tag = get32(req + RREC_MSG_OFF_TAG);
+	h->length = get32(req + RREC_MSG_OFF_LENGTH);
+	h->crc = get32(req + RREC_MSG_OFF_CRC);
+	if (get16(req + RREC_MSG_OFF_MAGIC) != RREC_MSG_MAGIC) {
+		return RREC_ST_BAD_MAGIC;                     /* FW l.690 */
+	}
+	if (h->length > RXF_MAX_REQ_PAYLOAD || h->length != req_len - RREC_MSG_HEADER_BYTES) {
+		return RREC_ST_BAD_ARG;
+	}
+	expect = rxf_proto_payload_crc(req + RREC_MSG_HEADER_BYTES, h->length);
+	if (h->crc != expect) {
+		return RREC_ST_BAD_CRC;                       /* FW l.690 */
+	}
+	if (!rxf_proto_opcode_known(h->opcode)) {
+		return RREC_ST_UNKNOWN_OPCODE;                /* FW l.689 */
+	}
+	return RREC_ST_OK;

+	uint32_t len;

+	return emit(user, out, RREC_MSG_HEADER_BYTES + len, true);

+/* Answer with the payload already at stage + 16 (or an error status). */
+static int finish_small(struct rxf_ctx *c, rxf_emit_fn emit, void *user, const struct rxf_req *h, enum rrec_status st,
+			uint32_t len, struct rxf_result *res)
+{
+	int rc;
+
+	if (stopped(c)) {
+		return RXF_ABORTED;                           /* FW l.179: no response for an aborted command */
+	}
+	res->status = st;
+	if (st != RREC_ST_OK) {
+		rc = emit_error(emit, user, h->opcode, st, h->tag);
+		return outcome_of(rc, c);
+	}
+	res->length = len;
+	rxf_proto_put_header(c->stage, h->opcode, st, h->tag, len, rxf_proto_payload_crc(c->stage + RREC_MSG_HEADER_BYTES, len));
+	rc = emit(user, c->stage, RREC_MSG_HEADER_BYTES + len, true);
+	return outcome_of(rc, c);

+static int read_range(struct rxf_ctx *c, uint32_t mps, const struct rxf_req *h, const uint8_t *p, rxf_emit_fn emit,
+		      void *user, struct rxf_result *res)
+{
+	uint32_t rec, first, count, crc = 0u, rem, total, n;
+	enum rrec_status st;
+	uint8_t *s = c->stage;
+
+	if (h->length != RREC_READ_REQ_BYTES) {
+		return finish_small(c, emit, user, h, RREC_ST_BAD_ARG, 0u, res);
+	}
+	rec = get32(p);
+	first = get32(p + 4);
+	count = get32(p + 8);
+	if (count < RREC_READ_COUNT_MIN || count > RREC_READ_COUNT_MAX) {              /* FW l.698 */
+		return finish_small(c, emit, user, h, RREC_ST_BAD_ARG, 0u, res);
+	}
+	if (first + count < first) {
+		return finish_small(c, emit, user, h, RREC_ST_NOT_FOUND, 0u, res);
+	}
+	/* every chunk must exist before any byte is sent (rxf.h range_check, FW l.661) */
+	st = c->be->range_check(rec, first, count);
+	if (st != RREC_ST_OK) {
+		return finish_small(c, emit, user, h, st, 0u, res);
+	}
+	/* pass 1: the header's CRC covers the whole payload and comes before it (FW l.686, rxf.h) */
+	for (uint32_t i = 0; i < count; i++) {
+		if (stopped(c)) {
+			return RXF_ABORTED;
+		}
+		st = c->be->read_chunk(rec, first + i, s);
+		if (st != RREC_ST_OK) {
+			return finish_small(c, emit, user, h, st, 0u, res);
+		}
+		c->crcs[i] = crc32_ieee(s, RREC_CHUNK_BYTES);
+		crc = crc32_ieee_update(crc, s, RREC_CHUNK_BYTES);
+	}
+	res->status = RREC_ST_OK;
+	res->length = count * RREC_CHUNK_BYTES;
+	rxf_proto_put_header(s, h->opcode, RREC_ST_OK, h->tag, res->length, crc);
+	if (mps == 0u) {
+		mps = RREC_USB_MPS_FS;
+	}
+	/* pass 2: header, then each chunk exactly as stored; the carry (less than one packet) stays in front of the
+	 * next chunk so that every piece but the last is a whole number of packets */
+	rem = RREC_MSG_HEADER_BYTES;
+	for (uint32_t i = 0; i < count; i++) {
+		if (stopped(c)) {
+			return RXF_ABORTED;
+		}
+		st = c->be->read_chunk(rec, first + i, s + rem);
+		if (st != RREC_ST_OK || crc32_ieee(s + rem, RREC_CHUNK_BYTES) != c->crcs[i]) {
+			res->status = RREC_ST_STORAGE_ERROR;  /* the chunk changed or failed after the header was fixed */
+			return RXF_SEND_FAILED;
+		}
+		total = rem + RREC_CHUNK_BYTES;
+		n = (i + 1u == count) ? total : (total / mps) * mps;
+		if (n != 0u) {
+			int rc = emit(user, s, n, i + 1u == count);
+
+			if (rc != 0) {
+				return outcome_of(rc, c);
+			}
+		}
+		rem = total - n;
+		memmove(s, s + n, rem);
+	}
+	return RXF_SENT;
+}
+
+static int dispatch(struct rxf_ctx *c, uint32_t mps, const struct rxf_req *h, const uint8_t *p, rxf_emit_fn emit,
+		    void *user, struct rxf_result *res)
+{
+	const struct rxf_backend *be = c->be;
+	uint8_t *out = c->stage + RREC_MSG_HEADER_BYTES;
+	enum rrec_status st = RREC_ST_OK;
+	uint32_t len = 0u, plen = h->length;
+	size_t olen = 0u;
+
+	switch (h->opcode) {
+	case RREC_OP_GET_INFO:                                                  /* FW l.696 */
+		if (plen != 0u) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->get_info(out);
+		len = RREC_GET_INFO_BYTES;
+		break;
+	case RREC_OP_LIST_RECORDINGS: {                                         /* FW l.697 */
+		uint16_t max;
+
+		if (plen != RREC_LIST_REQ_BYTES) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		max = get16(p + 4);
+		if (max < RREC_LIST_MAX_MIN || max > RREC_LIST_MAX_MAX) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->list(get32(p), max, out, &olen);
+		if (st == RREC_ST_OK && (olen < 2u || olen > RXF_LIST_PAYLOAD_MAX)) {
+			st = RREC_ST_STORAGE_ERROR;
+		}
+		len = (uint32_t)olen;
+		break;
+	}
+	case RREC_OP_READ_RANGE:                                                /* FW l.698 */
+		return read_range(c, mps, h, p, emit, user, res);
+	case RREC_OP_CONFIRM:                                                   /* FW l.699 */
+		if (plen != RREC_CONFIRM_REQ_BYTES) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->confirm(get32(p), get32(p + 4));
+		break;
+	case RREC_OP_DELETE: {                                                  /* FW l.700 */
+		uint32_t free_after = 0u;
+
+		if (plen != RREC_DELETE_REQ_BYTES) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->del(get32(p), &free_after);
+		put32(out, free_after);
+		len = RREC_DELETE_RESP_BYTES;
+		break;
+	}
+	case RREC_OP_SET_TIME:                                                  /* FW l.701 */
+		if (plen != RREC_SET_TIME_REQ_BYTES) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->set_time((int64_t)get64(p), (int16_t)get16(p + 8));
+		break;
+	case RREC_OP_GET_BATTERY:                                               /* FW l.702 */
+		if (plen != 0u) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->get_battery(out);
+		len = RREC_GET_BATTERY_BYTES;
+		break;
+	case RREC_OP_FW_BEGIN:                                                  /* FW l.703 */
+		if (plen != RREC_FW_BEGIN_REQ_BYTES) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->fw_begin(get32(p), get32(p + 4));
+		break;
+	case RREC_OP_FW_DATA:                                                   /* FW l.704 */
+		if (plen < 4u + RREC_FW_DATA_MIN || plen > 4u + RREC_FW_DATA_MAX) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->fw_data(get32(p), p + 4, plen - 4u);
+		break;
+	case RREC_OP_FW_END:                                                    /* FW l.705 */
+		if (plen != 0u) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->fw_end();
+		break;
+	case RREC_OP_GET_LOG: {                                                 /* FW l.706 */
+		uint16_t max;
+
+		if (plen != RREC_GET_LOG_REQ_BYTES) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		max = get16(p + 4);
+		if (max < RREC_GET_LOG_MAX_MIN || max > RREC_GET_LOG_MAX_MAX) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->get_log(get32(p), max, out, &olen);
+		if (st == RREC_ST_OK && (olen < 2u || olen > RXF_LOG_PAYLOAD_MAX)) {
+			st = RREC_ST_STORAGE_ERROR;
+		}
+		len = (uint32_t)olen;
+		break;
+	}
+	case RREC_OP_REBOOT:                                                    /* FW l.707 */
+		if (plen != 0u) {
+			st = RREC_ST_BAD_ARG;
+		}
+		break;                                                          /* the join reboots in after_response */
+	case RREC_OP_FACTORY:                                                   /* FW l.708 */
+		if (plen < 1u) {
+			st = RREC_ST_BAD_ARG;
+			break;
+		}
+		st = be->factory(p, plen, out, &olen);
+		if (st == RREC_ST_OK && olen > RXF_FACTORY_RESULT_MAX) {
+			st = RREC_ST_STORAGE_ERROR;
+		}
+		len = (uint32_t)olen;
+		break;
+	default:
+		st = RREC_ST_UNKNOWN_OPCODE;
+		break;
+	}
+	return finish_small(c, emit, user, h, st, len, res);
+}
+
+int rxf_proto_handle(struct rxf_ctx *c, uint32_t mps, const uint8_t *req, size_t req_len, rxf_emit_fn emit,
+		     void *user, struct rxf_result *res)
+{
+	struct rxf_req h;
+	enum rrec_status st;
+	int rc;
+
+	memset(res, 0, sizeof(*res));
+	if (!c->be->may_answer()) {                       /* FW l.673: CHARGING, TRANSFER, FACTORY_TEST only */
+		res->outcome = RXF_NO_ANSWER;
+		return RXF_NO_ANSWER;
+	}
+	st = rxf_proto_check(req, req_len, &h);
+	res->opcode = h.opcode;
+	res->tag = h.tag;
+	if (st != RREC_ST_OK) {
+		res->status = st;
+		rc = emit_error(emit, user, h.opcode, st, h.tag);
+		res->outcome = rc == 0 ? RXF_SENT : RXF_SEND_FAILED;
+		return res->outcome;
+	}
+	if (__atomic_exchange_n(&c->running, 1, __ATOMIC_ACQ_REL) != 0) {
+		res->status = RREC_ST_BUSY;               /* FW l.711 */
+		rc = emit_error(emit, user, h.opcode, RREC_ST_BUSY, h.tag);
+		res->outcome = rc == 0 ? RXF_SENT : RXF_SEND_FAILED;
+		return res->outcome;
+	}
+	__atomic_store_n(&c->aborted, 0, __ATOMIC_RELEASE);
+	res->dispatched = true;
+	res->outcome = (enum rxf_outcome)dispatch(c, mps, &h, req + RREC_MSG_HEADER_BYTES, emit, user, res);
+	__atomic_store_n(&c->running, 0, __ATOMIC_RELEASE);
+	if (res->outcome == RXF_SENT && c->be->after_response != NULL) {
+		c->be->after_response(h.opcode, res->status, h.tag, h.length);   /* FW l.705, l.707 */
+	}
+	return res->outcome;
+}

+/*
+ * rxf_proto.h: the pure part of the USB transfer (software step 6, unit U4). Not public: the logic suites and
+ * rxf_class.c include it. No kernel call (stdint, stdbool, string, and crc32_ieee() of the C library, decision D11a).
+ * "FW l.N" = line N of design/firmware.md. Numbers only from rrec_spec.h.

+ * What it does: validates one complete request (FW 6.2), checks the argument limits (FW 6.3) BEFORE it calls the
+ * backend, builds the response and hands it to an emit function in pieces whose sizes are multiples of the bus

+/* a TRANSFER with the card mounted, then `records` event log records queued on a card that stalls each write by
+ * `stall_ms`, then GET_LOG sent (not waited for) */
+static void transfer_with_busy_store(uint32_t stall_ms, int records)
+{
+	uint8_t pl[RREC_GET_LOG_REQ_BYTES];
+	uint8_t a[4] = { 0 };
+
+	rt_fresh(NULL);
+	rt_vbus(true);
+	rt_usb_configured();
+	zassert_equal(rt_state(), RC_ST_TRANSFER, "TRANSFER");
+	zassert_true(rt_sd(), "VDD_SD on in TRANSFER");
+	rhw_sim_disk_stall_ms(stall_ms);
+	for (int i = 0; i < records; i++) {
+		a[0] = (uint8_t)i;
+		rrec_log(RREC_LOG_BATTERY, a, sizeof(a));   /* records to write: the storage thread and GET_LOG write them */
+	}
+	sys_put_le32(1, pl);
+	sys_put_le16(RREC_GET_LOG_MAX_MAX, &pl[4]);
+	zassert_equal(rt_usb_send(RREC_OP_GET_LOG, pl, sizeof(pl)), 0, "GET_LOG sent");
+	rt_sleep(100);
+	zassert_true(rxf_busy(), "the command runs");
+	rt_steps_clear();
+	rhw_sim_log_clear();
+}

+/* the end of a case: the command ended (the transfer thread is free again) and the card no longer stalls */

+	}

+	rt_settle();
