#include "rt.h"

static uint64_t frame_hash(void)
{
	uint64_t h = 0xcbf29ce484222325ull;

	rscr_sync();
	rscr_capture_rgb(frame);
	for (size_t i = 0; i < sizeof(frame); i++) {
		h = (h ^ frame[i]) * 0x100000001b3ull;
	}
	return h;
}
